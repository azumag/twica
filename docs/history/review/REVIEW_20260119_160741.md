// IMPLEMENTED.md に記載の通り、sameSite: 'lax' を使用
cookieStore.set('csrf_token', token, {
  httpOnly: true,
  secure: process.env.NODE_ENV === 'production',

### 問題2: 楽観的ロックの再帰的リトライの無限ループリスク

**重大度**: 中

**詳細**:
IMPLEMENTED.md の `setCSRFToken()` 関数（行0116-01140）では、バージョン不一致時に再帰的に `setCSRFToken()` を再呼び出ししています：

```typescript
if (currentSession.version !== session.version) {
  logger.warn('CSRF token generation: Version mismatch, retrying', {
    userId,
    expectedVersion: session.version,
    actualVersion: currentSession.version,
  })
  return setCSRFToken()  // 再帰的にリトライ
}
```

**問題点**:
- 极端なケースで無限ループに陥る可能性がある
- リトライ回数に上限がない
- 同時リクエストが頻繁に発生する場合、パフォーマンスが低下する

**修正提案**:
```typescript
const MAX_RETRY_COUNT = 3

export async function setCSRFToken(retryCount: number = 0): Promise<string> {
  // ... 既存のコード ...

  if (currentSession.version !== session.version) {
    if (retryCount >= MAX_RETRY_COUNT) {
      logger.error('CSRF token generation: Max retry count exceeded', {
        userId,
        retryCount,
        expectedVersion: session.version,
        actualVersion: currentSession.version,
      })
      throw new Error('CSRF token generation failed: Concurrent modification detected')
    }
    
    logger.warn('CSRF token generation: Version mismatch, retrying', {
      userId,
      retryCount,
      expectedVersion: session.version,
      actualVersion: currentSession.version,
    })
    
    // 少し待ってからリトライ
    await new Promise(resolve => setTimeout(resolve, 10))
    return setCSRFToken(retryCount + 1)
  }
  // ...
}
```

### 問題3: CSRFトークン検証のロジックにおける潜在的な問題

**重大度**: 低

**詳細**:
`validateCSRFToken()` 関数では、cookieからトークンを取得する際に `csrf_token` cookieの存在を確認していますが、以下の点に注意が必要です：

```typescript
const requestToken = cookieStore.get('csrf_token')?.value

if (!requestToken) {
  logger.info('CSRF validation failed: CSRF token missing in request', {
    userId: session.twitchUserId,
    endpoint: request.url,
  })
  return { valid: false, error: ERROR_MESSAGES.CSRF_TOKEN_INVALID }
}
```

**問題点**:
- cookie名はハードコードされているが、定数として定義されていない（COOKIE_NAMESを使用すべき）
- cookieの取得失敗がどのような条件で発生するか документа化されていない

**修正提案**:
```typescript
// 定数を使用
const csrfTokenCookie = cookieStore.get(COOKIE_NAMES.CSRF_TOKEN)?.value

if (!csrfTokenCookie) {
  logger.info('CSRF validation failed: CSRF token cookie missing', {
    userId: session.twitchUserId,
    endpoint: request.url,
    hasSessionCookie: !!sessionCookie,
    hasCsrfTokenHash: !!sessionTokenHash,
  })
  return { valid: false, error: ERROR_MESSAGES.CSRF_TOKEN_INVALID }
}
```

### 問題4: クライアント側CSRFユーティリティの過剰な抽象化

**重大度**: 低

**詳細**:
IMPLEMENTED.md に記載されている `fetchWithCSRF()` 関数（行0230-0236）は、HttpOnly Cookieパターンでは単純な `fetch()` のラッパーになっています：

```typescript
export async function fetchWithCSRF(
  url: string,
  options: RequestInit = {}
): Promise<Response> {
  // CSRFトークンはhttpOnly cookieに保存され、自動的に送信される
  return fetch(url, options)
}
```

**問題点**:
- この関数は現在の実装では何もしない
- 将来的な変更を考慮して残していると思われるが、コードの意図が不明確
- 関数を維持する理由が документа化されていない

**修正提案**:

**選択肢A - 関数を削除**:
```typescript
// fetchWithCSRF関数を削除し、通常のfetchを使用
// 既存の呼び出し箇所:
const response = await fetch('/api/cards', {
  method: 'POST',
  headers: { 'Content-Type': 'application/json' },
  body: JSON.stringify(cardData),
})
```

**選択肢B - 関数を維持する場合、コメントを追加**:
```typescript
/**
 * CSRF保護されたfetchラッパー
 * 注: HttpOnly Cookieパターンでは、CSRFトークンは自動的に送信されるため、
 * 特別な処理は不要。将来的なパターン変更の可能性に備えて維持。
 */
export async function fetchWithCSRF(
  url: string,
  options: RequestInit = {}
): Promise<Response> {
  return fetch(url, options)
}
```

### 問題5: エラーハンドリングの詳細度の不足

**重大度**: 低

**詳細**:
`validateCSRFToken()` 関数のエラーハンドリングでは、一部のエラーが単純な文字列に変換されています：

```typescript
} catch (error) {
  logger.error('CSRF validation failed: Hash comparison error', {
    userId: session.twitchUserId,
    error: error instanceof Error ? error.message : String(error),
    stack: error instanceof Error ? error.stack : undefined,
  })
  return { valid: false, error: ERROR_MESSAGES.CSRF_TOKEN_INVALID }
}
```

**問題点**:
- `timingSafeEqual` のバッファ長不一致エラーはcaughtされるが、具体的なエラー種類が区別されていない
- エラーの種類によって異なる対応が必要可能性がある

**修正提案**:
```typescript
} catch (error) {
  if (error instanceof Error && error.name === 'OperationError') {
    // timingSafeEqualのバッファ長不一致エラー
    logger.warn('CSRF validation failed: Buffer length mismatch', {
      userId: session.twitchUserId,
      error: error.message,
    })
  } else {
    logger.error('CSRF validation failed: Unexpected error', {
      userId: session.twitchUserId,
      error: error instanceof Error ? error.message : String(error),
      stack: error instanceof Error ? error.stack : undefined,
    })
  }
  return { valid: false, error: ERROR_MESSAGES.CSRF_TOKEN_INVALID }
}
```

### 問題6: ログ出力のセキュリティ情報漏洩リスク

**重大度**: 低

**詳細**:
ログ出力でIPアドレスやユーザー情報が記録されています：

```typescript
logger.warn('CSRF token validation failed: Token mismatch (potential attack)', {
  userId: session.twitchUserId,
  ip: request.headers.get('x-forwarded-for'),
  referer: request.headers.get('referer'),
  endpoint: request.url,
})
```

**問題点**:
- PII（個人識別情報）がログに記録される
- GDPRやデータ保護規制への準拠に影響する可能性
- ログの保管と管理に関する方針が不明確

**修正提案**:
```typescript
logger.warn('CSRF token validation failed: Token mismatch (potential attack)', {
  userId: session.twitchUserId,  // ユーザー識別子は必要に応じて維持
  // IPとrefererはマスキングを検討
  ipHash: hashIp(request.headers.get('x-forwarded-for')),  // IPをハッシュ化
  // endpointはURLパラメータが含まれる可能性があるため注意
  endpoint: sanitizeEndpoint(request.url),
  timestamp: new Date().toISOString(),
})
```

### 問題7: セッションのバージョン管理に関するドキュメントの不足

**重大度**: 低

**詳細**:
楽観的ロックのためにセッションに `version` フィールドが追加されていますが、以下の点が不明確です：

- バージョンの初期値が `1` である理由
- バージョンが `Number.MAX_SAFE_INTEGER` を超えた場合の処理
- バージョンのインクリメントロジックが他のセッション更新処理と整合しているか

**修正提案**:
セッション管理に関するドキュメントを追加：

```typescript
/**
 * セッションの楽観的ロックバージョン
 * - 初期値: 1
 * - CSRFトークン更新ごとにインクリメント
 * - 最大値到達時は古いセッションを無効化して再作成
 */
const MAX_SESSION_VERSION = Number.MAX_SAFE_INTEGER - 1

function incrementSessionVersion(currentVersion: number): number {
  if (currentVersion >= MAX_SESSION_VERSION) {
    return 1  // 循環再用
  }
  return currentVersion + 1
}
```
