```typescript
export function parseSession(raw: string): Session {
  try {
    const parsed = JSON.parse(raw)
    if (!parsed.twitchUserId || !parsed.twitchUsername || !parsed.expiresAt) {
      throw new Error('Invalid session: missing required fields')
    }
    // バージョンフィールドがない場合、デフォルト値を設定
    if (typeof parsed.version !== 'number') {
      parsed.version = 1
    }
    return parsed as Session
  } catch (error) {
    if (error instanceof Error) {
      throw new Error(`Invalid session format: ${error.message}`)
    }
    throw new Error('Invalid session format')
  }
}
```

## セキュリティ上の懸念

### 7. HttpOnly Cookieパターンの根本的な問題

**場所**: src/lib/csrf.ts:133-211

**問題点**:
- HttpOnly Cookieパターンでは、Cookieが自動的に送信されるため、SameSite='lax'でのみCSRF保護が機能する
- SameSite='lax'はGETリクエストのトップレベルナビゲーションを許可し、クロスサイトPOSTリクエストをブロックする
- しかし、古いブラウザ（Safari < 12, IE）ではSameSiteがサポートされていないため、CSRF保護が機能しない

**実装コード**:
```typescript
// src/lib/csrf.ts:110-116
cookieStore.set(COOKIE_NAMES.CSRF_TOKEN, token, {
  httpOnly: true,
  secure: process.env.NODE_ENV === 'production',
  sameSite: 'lax',  // ← 古いブラウザではサポートされていない
  path: '/',
  maxAge: SESSION_CONFIG.MAX_AGE_SECONDS,
})
```

**影響**:
- 古いブラウザを使用しているユーザーは、CSRF攻撃に対して脆弱になる可能性がある
- 攻撃者は古いブラウザのユーザーをターゲットにしたCSRF攻撃を行う可能性がある

**修正案**:
1. ユーザーエージェントの検出を行い、古いブラウザに対して追加の保護を提供する
2. または、Refererヘッダーの検証を追加する

**推奨されるアクション**:
1. Refererヘッダーの検証を追加する
2. ユーザーエージェントの検出を行い、古いブラウザに対して追加の保護を提供する

```typescript
    const identifier = await getRateLimitIdentifier(request, session?.twitchUserId);
    const rateLimitResult = await checkRateLimit(rateLimits.authLogout, identifier);

    if (!rateLimitResult.success) {
      return NextResponse.redirect(`${process.env.NEXT_PUBLIC_APP_URL}/?error=${encodeURIComponent(ERROR_MESSAGES.RATE_LIMIT_EXCEEDED)}`)
    }

    const csrfValidation = await validateCSRFToken(request);
    if (!csrfValidation.valid) {
      return NextResponse.redirect(`${process.env.NEXT_PUBLIC_APP_URL}/?error=${encodeURIComponent(ERROR_MESSAGES.FORBIDDEN)}`);
    }

    if (session) {
      await deleteTwitchTokens(session.twitchUserId);
    }

    await clearCSRFToken()
    await clearSession()
    return NextResponse.redirect(`${process.env.NEXT_PUBLIC_APP_URL}/`)
  } catch (error) {
    return handleApiError(error, "Auth Logout API: POST")
  }
}
```

```typescript
// src/lib/session.ts の修正内容

export function parseSession(raw: string): Session {
  try {
    const parsed = JSON.parse(raw)
    if (!parsed.twitchUserId || !parsed.twitchUsername || !parsed.expiresAt) {
      throw new Error('Invalid session: missing required fields')
    }
    // バージョンフィールドがない場合、デフォルト値を設定
    if (typeof parsed.version !== 'number') {
      parsed.version = 1
    }
    return parsed as Session
  } catch (error) {
    if (error instanceof Error) {
      throw new Error(`Invalid session format: ${error.message}`)
    }
    throw new Error('Invalid session format')
  }
}
```
